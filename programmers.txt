# The programmers of arduino:avr are inherited (build.core=arduino:arduino) and shown by
# the IDE, so only the ones added for debugging are here: dw-link and the debug probes
# PyAvrOCD supports. Most of them need the avrdude 8.3 of this platform; arduino:avr's
# avrdude 6.3 does not know 11 of their protocol names.

# dw-link (version 7 and later) programs as a JTAGICE mkII in ISP mode over its serial
# port. avrdude's jtag2isp stays at the 19200 bps the mkII starts with: its open routine
# never sends PAR_BAUD_RATE (avrdude 8.3, stk500v2.c), so -b would be ignored.
dwlink.name=dw-link (ISP mode)
dwlink.communication=serial
dwlink.protocol=jtag2isp
dwlink.program.protocol=jtag2isp
dwlink.program.tool=avrdude
dwlink.program.tool.default=avrdude
dwlink.program.extra_params=-P{serial.port}
